🧾 Cheat sheet (copy & practice)
whoami
sudo whoami
hostname
hostname -I
uname
uname -r
uname -m
uname -a
echo $SHELL
echo "Hello, Linux"
cat /etc/os-release
whoami; hostname; uname -r
grep PRETTY /etc/os-release
